//Data Cleaning
function runDataQualityChecks() {

  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const rawSheet = ss.getSheetByName("Raw_Data");
  const qaSheet = ss.getSheetByName("QA_Results");
  const settingsSheet = ss.getSheetByName("Settings");

  const data = rawSheet.getDataRange().getValues();

  const headers = data[0];
  const rows = data.slice(1);

  // Map column names to their positions
  const col = {};
  headers.forEach((header, index) => {
    col[header] = index;
  });

  // Read approved values from Settings
  const validRegions = settingsSheet
    .getRange("B2").getValue()
    .split(",")
    .map(value => value.trim());

  const validStatuses = settingsSheet
    .getRange("B3").getValue()
    .split(",")
    .map(value => value.trim());

  // Store validation issues
  const issues = [];

  // Helper function to record an issue
  function addIssue(row, orderId, field, message, severity) {
    issues.push([
      row + 2,
      orderId,
      field,
      message,
      severity
    ]);
  }


  // Group records by Order ID
  const recordsByOrderId = new Map();

  rows.forEach((row, index) => {

    const id = String(row[col.Order_ID]).trim();

    if (id !== "") {

      if (!recordsByOrderId.has(id)) {
        recordsByOrderId.set(id, []);
      }

      recordsByOrderId.get(id).push({
        row: row,
        sourceRow: index + 2
      });

    }

  });

  // Store duplicate findings by source row
  const duplicateFindings = new Map();

  recordsByOrderId.forEach((records, id) => {

    // Only investigate IDs appearing more than once
    if (records.length <= 1) {
      return;
    }

    // Normalize surrounding whitespace before comparing records
    const normalizedRecords = records.map(record =>
      record.row.map(value =>
        typeof value === "string" ? value.trim() : value
      )
    );

    // Check whether all records are identical
    const firstRecord = JSON.stringify(normalizedRecords[0]);

    const allIdentical = normalizedRecords.every(
      record => JSON.stringify(record) === firstRecord
    );

    if (allIdentical) {

      // Keep the first occurrence; flag subsequent copies
      records.slice(1).forEach(record => {
        duplicateFindings.set(record.sourceRow, {
          message: "Exact duplicate; exclude this occurrence",
          severity: "Warning"
        });
      });

    } else {

      // Conflicting records: flag every occurrence as an error
      records.forEach(record => {
        duplicateFindings.set(record.sourceRow, {
          message: "Conflicting records share the same order ID",
          severity: "Error"
        });
      });

    }

  });


  // Validate each record
  rows.forEach((row, index) => {

    const sourceRow = index + 2;
    const orderId = String(row[col.Order_ID]).trim();

    // Required fields
    const requiredFields = [
      "Order_ID",
      "Customer_ID",
      "Product"
    ];

    requiredFields.forEach(field => {
      if (String(row[col[field]]).trim() === "") {
        addIssue(
          index,
          orderId,
          field,
          "Required value is missing",
          "Error"
        );
      }
    });

        // Duplicate Order ID
    const duplicateFinding = duplicateFindings.get(sourceRow);

    if (duplicateFinding) {
      addIssue(
        index,
        orderId,
        "Order_ID",
        duplicateFinding.message,
        duplicateFinding.severity
      );
    }

    // Validate date
    const dateValue = row[col.Order_Date];

    if (!isValidDateValue(dateValue)) {
      addIssue(
        index,
        orderId,
        "Order_Date",
        "Invalid date",
        "Error"
      );
    }

    // Validate region
    const region = String(row[col.Region]).trim();

    if (!validRegions.includes(region)) {
      addIssue(
        index,
        orderId,
        "Region",
        "Unapproved region",
        "Error"
      );
    }

    // Validate quantity
    const quantity = row[col.Quantity];

    if (
      typeof quantity !== "number" ||
      !Number.isFinite(quantity) ||
      quantity <= 0
    ) {
      addIssue(
        index,
        orderId,
        "Quantity",
        "Quantity must be a positive number",
        "Error"
      );
    }

    // Validate unit price
    const price = row[col.Unit_Price];

    if (
      typeof price !== "number" ||
      !Number.isFinite(price) ||
      price < 0
    ) {
      addIssue(
        index,
        orderId,
        "Unit_Price",
        "Price must be a non-negative number",
        "Error"
      );
    }

    // Validate status
    const status = String(row[col.Status]).trim();

    const matchingStatus = validStatuses.find(
      value => value.toLowerCase() === status.toLowerCase()
    );

    if (!matchingStatus) {
      addIssue(
        index,
        orderId,
        "Status",
        "Unapproved status",
        "Error"
      );

    } else if (status !== matchingStatus) {
      addIssue(
        index,
        orderId,
        "Status",
        "Inconsistent capitalization",
        "Warning"
      );
    }

  });

  // Write results to QA_Results
  qaSheet.clear();

  const outputHeaders = [
    "Source Row",
    "Order ID",
    "Field",
    "Issue",
    "Severity"
  ];

  qaSheet.getRange(1, 1, 1, outputHeaders.length)
    .setValues([outputHeaders]);

  if (issues.length > 0) {
    qaSheet.getRange(
      2, 1, issues.length, outputHeaders.length
    ).setValues(issues);
  }

  qaSheet.getRange(1, 1, 1, outputHeaders.length)
    .setFontWeight("bold")
    .setBackground("#cfe2f3");

  qaSheet.setFrozenRows(1);
  qaSheet.autoResizeColumns(1, outputHeaders.length);

  Logger.log(
    "Data quality checks complete. Issues found: " +
    issues.length
  );
}


// Helper function to validate dates
function isValidDateValue(value) {

  // Check if value is an actual Date object
  if (value instanceof Date) {
    return !isNaN(value.getTime());
  }

  // Check ISO date format: YYYY-MM-DD
  if (typeof value !== "string") {
    return false;
  }

  const match = value.match(
    /^(\d{4})-(\d{2})-(\d{2})$/
  );

  if (!match) {
    return false;
  }

  const year = Number(match[1]);
  const month = Number(match[2]);
  const day = Number(match[3]);

  const date = new Date(Date.UTC(year, month - 1, day));

  return (
    date.getUTCFullYear() === year &&
    date.getUTCMonth() === month - 1 &&
    date.getUTCDate() === day
  );
}

function buildCleanData() {

  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const rawSheet = ss.getSheetByName("Raw_Data");
  const qaSheet = ss.getSheetByName("QA_Results");
  const cleanSheet = ss.getSheetByName("Clean_Data");
  const settingsSheet = ss.getSheetByName("Settings");

  // Read raw data and QA results
  const rawData = rawSheet.getDataRange().getValues();
  const qaData = qaSheet.getDataRange().getValues();

  const headers = rawData[0];
  const rows = rawData.slice(1);

  // Map column names to their positions
  const col = {};

  headers.forEach((header, index) => {
    col[header] = index;
  });

  // Read approved statuses
  const validStatuses = settingsSheet
    .getRange("B3")
    .getValue()
    .split(",")
    .map(value => value.trim());

  // Collect source rows that should be excluded
  const excludedRows = new Set();

  qaData.slice(1).forEach(qaRow => {

    const sourceRow = Number(qaRow[0]);
    const issueDescription = String(qaRow[3]).trim();
    const severity = String(qaRow[4]).trim();

    if (
      severity === "Error" ||
      issueDescription === "Exact duplicate; exclude this occurrence"
    ) {
      excludedRows.add(sourceRow);
    }

  });

  // Build the cleaned dataset
  const cleanRows = [];

  rows.forEach((row, index) => {

    const sourceRow = index + 2;

    // Skip records that should be excluded
    if (excludedRows.has(sourceRow)) {
      return;
    }

    // Make a copy so Raw_Data remains unchanged
    const cleanRow = [...row];

    // Standardize status capitalization
    const status = String(cleanRow[col.Status]).trim();

    const matchingStatus = validStatuses.find(
      value => value.toLowerCase() === status.toLowerCase()
    );

    if (matchingStatus) {
      cleanRow[col.Status] = matchingStatus;
    }

    cleanRows.push(cleanRow);

  });

  // Clear previous cleaned data
  cleanSheet.clearContents();

  // Write headers
  cleanSheet.getRange(1, 1, 1, headers.length)
    .setValues([headers]);

  // Write cleaned records
  if (cleanRows.length > 0) {
    cleanSheet.getRange(
      2, 1, cleanRows.length, headers.length
    ).setValues(cleanRows);
  }

  // Format the output
  cleanSheet.getRange(1, 1, 1, headers.length)
    .setFontWeight("bold")
    .setBackground("#d9ead3");

  cleanSheet.setFrozenRows(1);
  cleanSheet.autoResizeColumns(1, headers.length);

  Logger.log(
    "Cleaning complete. Records kept: " + cleanRows.length +
    ". Records excluded: " + (rows.length - cleanRows.length)
  );

}

function buildSummaryMetrics() {

  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const cleanSheet = ss.getSheetByName("Clean_Data");
  const metricsSheet = ss.getSheetByName("Metrics");

  // Read cleaned data
  const data = cleanSheet.getDataRange().getValues();

  const headers = data[0];
  const rows = data.slice(1);

  // Map column names to positions
  const col = {};

  headers.forEach((header, index) => {
    col[header] = index;
  });

  // Initialize summary metrics
  let totalOrders = 0;
  let completedOrders = 0;
  let pendingOrders = 0;
  let cancelledOrders = 0;

  let completedRevenue = 0;
  let pendingRevenue = 0;
  let cancelledRevenue = 0;

  // Calculate metrics
  rows.forEach(row => {

    totalOrders++;

    const status = String(row[col.Status]).trim().toLowerCase();

    const quantity = row[col.Quantity];
    const unitPrice = row[col.Unit_Price];

    const orderValue = quantity * unitPrice;

    if (status === "complete") {

      completedOrders++;
      completedRevenue += orderValue;

    } else if (status === "pending") {

      pendingOrders++;
      pendingRevenue += orderValue;

    } else if (status === "cancelled") {

      cancelledOrders++;
      cancelledRevenue += orderValue;
    }

  });

  // Calculate average completed order value
  const averageOrderValue = completedOrders > 0
    ? completedRevenue / completedOrders
    : 0;

  // Prepare output
  const output = [
    ["Metric", "Value"],
    ["Total Orders", totalOrders],
    ["Completed Orders", completedOrders],
    ["Pending Orders", pendingOrders],
    ["Cancelled Orders", cancelledOrders],
    ["Completed Revenue", completedRevenue],
    ["Pending Revenue", pendingRevenue],
    ["Cancelled Revenue", cancelledRevenue],
    ["Average Completed Order Value", averageOrderValue]
  ];

  // Clear previous metrics
  metricsSheet.clearContents();

  // Write results
  metricsSheet.getRange(
    1, 1, output.length, output[0].length
  ).setValues(output);

  // Format headers
  metricsSheet.getRange("A1:B1")
    .setFontWeight("bold")
    .setBackground("#cfe2f3");

  // Format currency values
  metricsSheet.getRange("B6:B8")
    .setNumberFormat("$#,##0.00");

  // Format order counts
  metricsSheet.getRange("B2:B5")
    .setNumberFormat("0");

  metricsSheet.setFrozenRows(1);
  metricsSheet.autoResizeColumns(1, 2);

  Logger.log("Summary metrics calculated successfully.");

}

function buildBreakdowns() {

  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const cleanSheet = ss.getSheetByName("Clean_Data");
  const metricsSheet = ss.getSheetByName("Metrics");

  // Read cleaned data
  const data = cleanSheet.getDataRange().getValues();

  const headers = data[0];
  const rows = data.slice(1);

  // Map column names to positions
  const col = {};

  headers.forEach((header, index) => {
    col[header] = index;
  });

  // Helper function to create a breakdown
  function calculateBreakdown(groupField) {

    const results = new Map();

    rows.forEach(row => {

      const groupName = String(row[col[groupField]]).trim();
      const status = String(row[col.Status]).trim().toLowerCase();

      const quantity = row[col.Quantity];
      const unitPrice = row[col.Unit_Price];

      const orderValue = quantity * unitPrice;

      // Initialize a group if it doesn't exist
      if (!results.has(groupName)) {
        results.set(groupName, {
          completedOrders: 0,
          completedRevenue: 0,
          pendingOrders: 0,
          pendingRevenue: 0
        });
      }

      const group = results.get(groupName);

      // Accumulate metrics by status
      if (status === "complete") {

        group.completedOrders++;
        group.completedRevenue += orderValue;

      } else if (status === "pending") {

        group.pendingOrders++;
        group.pendingRevenue += orderValue;

      }

    });

    return results;
  }

  // Calculate product and regional breakdowns
  const productResults = calculateBreakdown("Product");
  const regionResults = calculateBreakdown("Region");

  // Convert results into spreadsheet-ready arrays
  function prepareOutput(results, groupHeader) {

    const output = [[
      groupHeader,
      "Completed Orders",
      "Completed Revenue",
      "Pending Orders",
      "Pending Revenue"
    ]];

    const sortedGroups = [...results.keys()].sort();

    sortedGroups.forEach(name => {

      const values = results.get(name);

      output.push([
        name,
        values.completedOrders,
        values.completedRevenue,
        values.pendingOrders,
        values.pendingRevenue
      ]);

    });

    return output;
  }

  const productOutput = prepareOutput(
    productResults,
    "Product"
  );

  const regionOutput = prepareOutput(
    regionResults,
    "Region"
  );

  // Clear previous breakdowns without touching summary metrics
  metricsSheet.getRange("A11:E100").clearContent();
  metricsSheet.getRange("G11:K100").clearContent();

  // Write product breakdown
  metricsSheet.getRange(
    11, 1,
    productOutput.length,
    productOutput[0].length
  ).setValues(productOutput);

  // Write regional breakdown
  metricsSheet.getRange(
    11, 7,
    regionOutput.length,
    regionOutput[0].length
  ).setValues(regionOutput);

  // Format table headers
  metricsSheet.getRange("A11:E11")
    .setFontWeight("bold")
    .setBackground("#cfe2f3");

  metricsSheet.getRange("G11:K11")
    .setFontWeight("bold")
    .setBackground("#cfe2f3");

  // Add section titles
  metricsSheet.getRange("A10")
    .setValue("Product Breakdown")
    .setFontWeight("bold");

  metricsSheet.getRange("G10")
    .setValue("Regional Breakdown")
    .setFontWeight("bold");

  // Format revenue columns
  metricsSheet.getRange("C12:C100")
    .setNumberFormat("$#,##0.00");

  metricsSheet.getRange("E12:E100")
    .setNumberFormat("$#,##0.00");

  metricsSheet.getRange("I12:I100")
    .setNumberFormat("$#,##0.00");

  metricsSheet.getRange("K12:K100")
    .setNumberFormat("$#,##0.00");

  // Format order counts
  metricsSheet.getRange("B12:B100")
    .setNumberFormat("0");

  metricsSheet.getRange("D12:D100")
    .setNumberFormat("0");

  metricsSheet.getRange("H12:H100")
    .setNumberFormat("0");

  metricsSheet.getRange("J12:J100")
    .setNumberFormat("0");

  metricsSheet.autoResizeColumns(1, 11);

  Logger.log("Product and regional breakdowns calculated.");

}

function buildProductRevenueChart() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const metricsSheet = ss.getSheetByName("Metrics");
  const dashboard = ss.getSheetByName("Dashboard");

  if (!metricsSheet || !dashboard) {
    throw new Error("Could not find Metrics or Dashboard sheet.");
  }

  // Remove an existing chart at the same position.
  dashboard.getCharts().forEach(chart => {
    const position = chart.getContainerInfo();

    if (
      position.getAnchorRow() === 14 &&
      position.getAnchorColumn() === 1
    ) {
      dashboard.removeChart(chart);
    }
  });

  // Step 1: Read the product breakdown dynamically.
  const lastMetricsRow = metricsSheet.getLastRow();

  const productData = metricsSheet.getRange(
    12, 1, lastMetricsRow - 11, 5
  ).getValues();

  const chartRows = [
    ["Product", "Completed Revenue", "Pending Revenue"]
  ];

  // Step 2: Add each product to the chart data.
  for (let i = 0; i < productData.length; i++) {
    const row = productData[i];

    // Stop when we reach the end of the product list.
    if (row[0] === "" || row[0] === null) {
      break;
    }

    chartRows.push([
      row[0],
      row[2],
      row[4]
    ]);
  }

  // Step 3: Create a helper range sized to the actual data.
  const chartData = dashboard.getRange(
    1, 14, chartRows.length, 3
  );

  chartData.setValues(chartRows);

  // Step 4: Build the chart.
  const chart = dashboard.newChart()
    .setChartType(Charts.ChartType.BAR)
    .addRange(chartData)
    .setNumHeaders(1)
    .setOption("useFirstRowAsHeaders", true)
    .setOption("useFirstColumnAsDomain", true)
    .setPosition(14, 1, 0, 0)
    .setOption("width", 600)
    .setOption("height", 340)

function buildRegionalRevenueChart() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const metricsSheet = ss.getSheetByName("Metrics");
  const dashboard = ss.getSheetByName("Dashboard");

  // Read regional data dynamically.
const lastMetricsRow = metricsSheet.getLastRow();

const regionalData = metricsSheet.getRange(
  12, 7, lastMetricsRow - 11, 5
).getValues();

  // Create a helper table.
  const chartData = dashboard.getRange("R1:T5");

  const chartRows = [
    ["Region", "Completed Revenue", "Pending Revenue"]
  ];

  for (let i = 0; i < regionalData.length; i++) {
  const row = regionalData[i];

  if (row[0] === "" || row[0] === null) {
    break;
  }

  chartRows.push([
    row[0],
    row[2],
    row[4]
  ]);
}
  chartData.setValues(chartRows);

  // Create the chart using our reusable function.
  createRevenueChart(
    "Revenue by Region",
    chartData,
    14,
    8
  );
}

function buildDataQualitySummary() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const rawSheet = ss.getSheetByName("Raw_Data");
  const cleanSheet = ss.getSheetByName("Clean_Data");
  const qaSheet = ss.getSheetByName("QA_Results");
  const metricsSheet = ss.getSheetByName("Metrics");

  const recordsReceived = rawSheet.getLastRow() - 1;
  const recordsRetained = cleanSheet.getLastRow() - 1;
  const recordsExcluded = recordsReceived - recordsRetained;
  const qaFindings = qaSheet.getLastRow() - 1;
  const retentionRate = recordsRetained / recordsReceived;

  const summary = [
    ["Data Quality Summary", "Value"],
    ["Records Received", recordsReceived],
    ["Records Retained", recordsRetained],
    ["Records Excluded", recordsExcluded],
    ["QA Findings", qaFindings],
    ["Retention Rate", retentionRate]
  ];

  metricsSheet.getRange("M1:N6").setValues(summary);

  metricsSheet.getRange("N6").setNumberFormat("0.0%");
}

function buildDashboardDataQualitySummary() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const metricsSheet = ss.getSheetByName("Metrics");
  const dashboard = ss.getSheetByName("Dashboard");

  // Read the data quality summary from Metrics
  const summary = metricsSheet.getRange("M2:N6").getValues();

  // Add a title
  dashboard.getRange("J32:K32").merge();
  dashboard.getRange("J32")
    .setValue("DATA QUALITY")
    .setBackground("#172A3A")
    .setFontColor("#FFFFFF")
    .setFontWeight("bold")
    .setHorizontalAlignment("center");

  // Display the summary values
  const displayRange = dashboard.getRange("J33:K37");
  displayRange.setValues(summary);

  // Format the panel
  displayRange
    .setBackground("#F1F5F5")
    .setFontColor("#172A3A")
    .setVerticalAlignment("middle");

  dashboard.getRange("J33:J37").setFontWeight("bold");
  dashboard.getRange("K33:K37").setHorizontalAlignment("center");
  dashboard.getRange("K37")
    .setNumberFormat("0.0%")
    .setHorizontalAlignment("center");

  // Add borders around the panel
  dashboard.getRange("J33:K37")
    .setBorder(true, true, true, true, true, true, "#D5DEDF",
      SpreadsheetApp.BorderStyle.SOLID);

  dashboard.setColumnWidth(10, 170);
  dashboard.setColumnWidth(11, 110);
}
